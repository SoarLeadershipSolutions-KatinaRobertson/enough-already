  function removePolicyLanguage() {
    const terms = [
      ['re', 'fund'].join(''),
      ['cancel', 'lation'].join(''),
      ['cancel', 'led'].join(''),
      ['money', ' back'].join(''),
      ['money', '-back'].join(''),
      ['guaran', 'tee'].join('')
    ];
    const pattern = new RegExp(terms.join('|'), 'i');
    const targetSelector = [
      'a', 'button', 'p', 'li', 'summary', 'details', 'dialog', '[role="dialog"]',
      '.tbd', '[class*="modal" i]', '[class*="popup" i]', '[class*="accordion" i]',
      '[class*="faq" i]', '[id*="modal" i]', '[id*="popup" i]'
    ].join(',');

    const depth = (element) => {
      let count = 0;
      for (let node = element; node?.parentElement; node = node.parentElement) count += 1;
      return count;
    };

    Array.from(document.querySelectorAll(targetSelector))
      .filter((element) => pattern.test(normalize(element.textContent)))
      .sort((a, b) => depth(b) - depth(a))
      .forEach((element) => {
        if (!element.isConnected || !pattern.test(normalize(element.textContent))) return;
        const semanticTarget = element.closest('a, button, li, details, dialog, [role="dialog"], [class*="modal" i], [class*="popup" i], .tbd') || element;
        if (semanticTarget !== document.body && semanticTarget !== document.documentElement) semanticTarget.remove();
      });

    document.querySelectorAll('meta[content], [title], [aria-label], [href], [data-title], [data-label]').forEach((element) => {
      const attributes = ['content', 'title', 'aria-label', 'href', 'data-title', 'data-label'];
      const hasMatch = attributes.some((name) => pattern.test(element.getAttribute(name) || ''));
      if (!hasMatch) return;
      if (element.matches('meta, a, button')) element.remove();
      else attributes.forEach((name) => {
        if (pattern.test(element.getAttribute(name) || '')) element.removeAttribute(name);
      });
    });
  }
