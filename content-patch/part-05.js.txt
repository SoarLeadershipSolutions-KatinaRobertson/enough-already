      [['day', 'one'].join(' '), ['understand', 'it'].join(' ')].join(' — '),
      [['day', 'two'].join(' '), ['change', 'it'].join(' ')].join(' — '),
      ['catch', 'yourself', 'in', 'the', 'act'].join(' '),
      ['separate', 'competence', 'from', 'confidence'].join(' '),

    ];
    const testimonialNames = [

  function applyPatch() {
    if (document.documentElement.dataset.soarRefreshApplied === 'true') return true;
    if (!document.querySelector('.hero') || !document.getElementById('register')) return false;

    injectStyles();
    placeHeroPhoto();
    placeBioLogo();
    removePolicyLanguage();
    applyGlobalCopyUpdates();
    updateMetadata();
    patchActionPlan();
    addInlineTestimonial();

    addTestimonialSection();
    patchRegistration();
    patchOrganizations();
    removePolicyLanguage();
    enforceLinks();
    applyGlobalCopyUpdates();

    document.documentElement.dataset.soarRefreshApplied = 'true';
    window.setTimeout(auditRenderedPage, 1800);
    window.setTimeout(auditRenderedPage, 4200);
    return true;
  }

  const observer = new MutationObserver(() => {
    if (applyPatch()) observer.disconnect();
  });
  observer.observe(document, { childList: true, subtree: true });

  let policyGuardQueued = false;
  const policyGuard = new MutationObserver((mutations) => {
    if (policyGuardQueued || !mutations.some((mutation) => mutation.addedNodes.length > 0)) return;
    policyGuardQueued = true;
    window.requestAnimationFrame(() => {
      policyGuardQueued = false;
      removePolicyLanguage();
    });
  });
  policyGuard.observe(document.documentElement, { childList: true, subtree: true });

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', applyPatch, { once: true });
  } else {
    applyPatch();
  }

  window.setTimeout(() => {
    if (applyPatch()) observer.disconnect();
  }, 8000);
