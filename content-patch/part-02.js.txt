    if (element instanceof HTMLImageElement) return element.naturalWidth * element.naturalHeight;
    const viewBox = element.getAttribute?.('viewBox')?.trim().split(/[\s,]+/).map(Number);
    if (viewBox?.length === 4 && viewBox.every(Number.isFinite)) return Math.abs(viewBox[2] * viewBox[3]);
    return 0;
  }

  function findHeroVisual(hero) {
    const heroRect = hero.getBoundingClientRect();
    const candidates = Array.from(hero.querySelectorAll('img, svg, canvas')).filter((element) => {
      if (element.matches('[data-soar-role]')) return false;
      if (element.closest('nav, header, a, .nav, .navbar, [class*="logo" i], [id*="logo" i]')) return false;
      const identity = normalize([
        element.getAttribute('alt'),
        element.getAttribute('aria-label'),
        element.id,
        element.className?.baseVal || element.className
      ].join(' ')).toLowerCase();
      if (/logo|wordmark|brand/.test(identity)) return false;
      const rect = element.getBoundingClientRect();
      const isSubstantial = (rect.width >= 150 && rect.height >= 150) || renderedArea(element) >= 40000;
      if (!isSubstantial) return false;
      const onVisualSide = heroRect.width < 760 || rect.left + rect.width / 2 >= heroRect.left + heroRect.width * .42;
      return onVisualSide;
    });
    return candidates.sort((a, b) => renderedArea(b) - renderedArea(a))[0] || null;
  }

  function visualShellFor(visual, hero) {
    let shell = visual.parentElement;
    let best = shell;
    while (shell?.parentElement && shell.parentElement !== hero) {
      const parent = shell.parentElement;
      const textLength = normalize(parent.textContent).length;
      const containsPrimaryContent = Boolean(parent.querySelector('h1, form, input, textarea'));
      if (containsPrimaryContent || textLength > 140) break;
      if (parent.children.length <= 6) best = parent;
      shell = parent;
    }
    return best || visual.parentElement;
  }

  function fallbackHeroShell(hero) {
    const heading = hero.querySelector('h1');
    if (!heading) {
      const shell = makeElement('div', 'soar-hero-photo-shell');
      hero.appendChild(shell);
      return shell;
    }
    let node = heading.parentElement;
    let layout = null;
    while (node?.parentElement && node.parentElement !== hero) {
      const parent = node.parentElement;
      if (parent.children.length >= 2 && parent.getBoundingClientRect().width >= 640) {
        layout = parent;
        break;
      }
      node = parent;
    }
    const shell = makeElement('div', 'soar-hero-photo-shell');
    (layout || hero).appendChild(shell);
    return shell;
  }

  function placeHeroPhoto() {
    const hero = document.querySelector('.hero');
    if (!hero) return;
    hero.classList.add('soar-brand-hero');

    let image = hero.querySelector('[data-soar-role="hero-photo"]');
    if (image) {
      image.setAttribute('src', HERO_IMAGE_PATH);
      return;
    }

    const visual = findHeroVisual(hero);
    const shell = visual ? visualShellFor(visual, hero) : fallbackHeroShell(hero);
    if (!shell) return;

    shell.replaceChildren();
    shell.className = `${shell.className || ''} soar-hero-photo-shell`.trim();
    shell.setAttribute('data-soar-photo-shell', 'hero');

    image = document.createElement('img');
    image.className = 'soar-hero-photo';
    image.setAttribute('data-soar-role', 'hero-photo');
    image.src = HERO_IMAGE_PATH;
    image.alt = 'Katina Christmas Robertson, leadership development expert and executive coach';
    image.width = 640;
    image.height = 800;
    image.loading = 'eager';
    image.decoding = 'async';
    image.fetchPriority = 'high';
    shell.appendChild(image);
  }

  function bioCopyColumn(marker, section) {
    let column = marker.parentElement;
    while (column?.parentElement && column.parentElement !== section) {
      const parent = column.parentElement;
      const style = getComputedStyle(parent);
      const isSectionLayout = parent.classList.contains('w') || parent.classList.contains('g2') || (style.display === 'grid' && parent.children.length > 1);
      if (isSectionLayout) break;
      if (parent.querySelector('p') && normalize(parent.textContent).length < 6000) column = parent;
      else break;
    }
    return column || marker.parentElement;
  }

  function placeBioLogo() {
    if (document.querySelector('[data-soar-role="bio-logo"]')) return;
    const marker = Array.from(document.querySelectorAll('h1, h2, h3')).find((heading) => {
      const value = normalize(heading.textContent).toLowerCase();
      return value.includes('katina christmas robertson') || value.includes('meet katina');
    });
    const section = sectionFor(marker);
    if (!marker || !section || section.classList.contains('hero')) return;

    section.classList.add('soar-bio-section');
    const column = bioCopyColumn(marker, section);
    if (!column) return;

    const lockup = makeElement('div', 'soar-bio-logo-lockup');
    lockup.setAttribute('aria-label', 'SOAR Leadership Solutions');
    const image = document.createElement('img');
    image.className = 'soar-bio-logo';
    image.setAttribute('data-soar-role', 'bio-logo');
    image.src = BIO_LOGO_PATH;
    image.alt = 'SOAR Leadership Solutions — Empowering Leaders. Elevating Teams. Driving Results.';
    image.width = 600;
    image.height = 450;
    image.loading = 'eager';
    image.decoding = 'async';
    lockup.appendChild(image);
    column.appendChild(lockup);
  }
