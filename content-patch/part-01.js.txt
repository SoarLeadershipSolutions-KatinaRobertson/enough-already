  const textIncludes = (element, phrase) => normalize(element?.textContent).toLowerCase().includes(phrase.toLowerCase());

  function findOne(selector, phrase) {
    return Array.from(document.querySelectorAll(selector)).find((element) => textIncludes(element, phrase)) || null;
  }

  function makeElement(tagName, className, html) {
    const element = document.createElement(tagName);

    if (html !== undefined) element.innerHTML = html;

  function updateTextNodes(root, replacer) {
    const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
    const nodes = [];
    while (walker.nextNode()) nodes.push(walker.currentNode);
    nodes.forEach((node) => {
      const replacement = replacer(node.nodeValue);
      if (replacement !== node.nodeValue) node.nodeValue = replacement;
    });
  }
